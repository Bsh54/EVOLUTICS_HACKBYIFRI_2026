from openai import OpenAI

client = OpenAI(
    api_key="sk-afri-...",
    base_url="https://build.lewisnote.com/v1"
)

response = client.chat.completions.create(
    model="gpt-5.4-nano",
    messages=[{"role": "user", "content": "Explique ce concept simplement"}]
)

print(response.choices[0].message.content)
